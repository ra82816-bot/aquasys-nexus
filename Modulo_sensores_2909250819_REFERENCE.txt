#define PH_SENSOR_PIN 34

#define TDS_SENSOR_PIN 35

#define DHT_PIN 15

#define DHT_TYPE DHT22

#define OLED_WIDTH 128

#define OLED_HEIGHT 64

#define OLED_RESET -1

#define OLED_ADDRESS 0x3C

#define BUTTON_UP 32

#define BUTTON_DOWN 33

#define BUTTON_SELECT 25

#define BUTTON_BACK 26



// --- OBJETOS ---

Adafruit_SSD1306 display(OLED_WIDTH, OLED_HEIGHT, &Wire, OLED_RESET);

Preferences preferences;

OneWire oneWire(ONE_WIRE_BUS);

DallasTemperature ds18b20(&oneWire);

DHT dht(DHT_PIN, DHT_TYPE);

WebServer server(80);

WiFiClientSecure espClient;

PubSubClient mqttClient(espClient);



// --- CALIBRACOES ---

float cal_ph7_voltage = 2.52;

float cal_ph4_voltage = 3.29;

float ph_slope, ph_intercept;

float calibration_low_raw = 645.0;

float calibration_high_raw = 2850.0;

float calibration_low_ec = 360.0;

float calibration_high_ec = 4588.0;



// --- VARIAVEIS ---

float temperature_C = 25.0;

float humidity = 0.0;

float water_temperature_C = 0.0;



enum Page {

  SHOW_PH,

  SHOW_TDS,

  SHOW_AIR_TEMP,

  SHOW_WATER_TEMP,

  SHOW_HUMIDITY,

  PAGE_COUNT

};

Page currentPage = SHOW_PH;



unsigned long lastDebounce[4] = {0,0,0,0};

const unsigned long debounceDelay = 200;

String ssid_sta = "";

String password_sta = "";

bool wifiConfigured = false;



// --- MQTT ---

#define MQTT_BROKER "8cda72f06f464778bc53751d7cc88ac2.s1.eu.hivemq.cloud"

#define MQTT_PORT 8883

#define MQTT_USERNAME "esp32-user"

#define MQTT_PASSWORD "HydroSmart123"

#define MQTT_TOPIC "aquasys/sensors/all"



unsigned long lastMQTTPublish = 0;

const unsigned long mqttInterval = 10000; // Publicar a cada 10 segundos



// --- PROTOTIPOS ---

void handleButtons();

void setup() {

  Serial.begin(115200);

  delay(1000);

  // --- PINOS BOTAO ---

  pinMode(BUTTON_UP, INPUT_PULLUP);

  pinMode(BUTTON_DOWN, INPUT_PULLUP);

  pinMode(BUTTON_SELECT, INPUT_PULLUP);

  pinMode(BUTTON_BACK, INPUT_PULLUP);

  // --- OLED ---

  if (!display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDRESS)) {

    Serial.println("Erro inicializando OLED");

    while (true);

  }

  display.clearDisplay();

  display.setTextSize(1);

  display.setTextColor(SSD1306_WHITE);

  // --- SENSORES ---

  dht.begin();

  ds18b20.begin();

  // --- CALIBRACOES ---

  preferences.begin("calib", false);

  loadCalibration();

  calculatePHCoefficients();

  // --- WIFI ---

  preferences.begin("wifi", false);

  ssid_sta = preferences.getString("ssid", "");

  password_sta = preferences.getString("password", "");

  if (ssid_sta.length() > 0) {

    displayMessage("Conectando WiFi...");

    connectToWiFi();

  }

  if (WiFi.status() != WL_CONNECTED) {

    startAPMode();

  } else {

    wifiConfigured = true;

    String msg = "WiFi conectado\nIP: " + WiFi.localIP().toString();

    displayMessage(msg.c_str());

    delay(2000);

  }

  // Configura as rotas do servidor web

  setupWebServer();

  // --- MQTT ---

  espClient.setInsecure();  // Para TLS sem certificado

  mqttClient.setServer(MQTT_BROKER, MQTT_PORT);

  displayMessage("Setup completo\nInicie conexao");

  delay(2000);

}



void loop() {

  if (!wifiConfigured && ssid_sta.length() > 0) {

    connectToWiFi();

  }

  if (WiFi.status() == WL_CONNECTED) {

    wifiConfigured = true;

    if (!mqttClient.connected()) {

      reconnectMQTT();

    }

    mqttClient.loop();



    unsigned long now = millis();

    if (now - lastMQTTPublish > mqttInterval) {

      lastMQTTPublish = now;

      publishData(

        voltageToPH(readAverageADC(PH_SENSOR_PIN) * 3.3 / 4095.0),

        temperatureCompensateEC(interpolateEC(readAverageADC(TDS_SENSOR_PIN)), temperature_C),

        temperature_C,

        humidity,

        water_temperature_C

      );

    }

  } else {

    server.handleClient(); // Se não conectado, atende portal configuração

  }

  readSensors();

  handleButtons();

  updateDisplay();

  delay(500);

}



void reconnectMQTT() {

  while (!mqttClient.connected()) {

    Serial.print("Conectando MQTT...");

    if (mqttClient.connect("ESP32_AquaSys", MQTT_USERNAME, MQTT_PASSWORD)) {

      Serial.println("Conectado ao broker MQTT!");

    } else {

      Serial.print("Falha, rc=");

      Serial.print(mqttClient.state());

      Serial.println(" Tentando novamente em 5s");

      delay(5000);

    }

  }

}



void publishData(float ph, float ec, float airTemp, float hum, float waterTemp) {

  StaticJsonDocument<256> doc;

  doc["ph"] = ph;

  doc["ec"] = ec;

  doc["airTemp"] = airTemp;

  doc["humidity"] = hum;

  doc["waterTemp"] = waterTemp;



  char buffer[256];

  size_t n = serializeJson(doc, buffer);

 

  mqttClient.publish(MQTT_TOPIC, buffer, n);

}



void connectToWiFi() {

  Serial.println("Tentando conectar WiFi: " + ssid_sta);

  WiFi.mode(WIFI_STA);

  WiFi.begin(ssid_sta.c_str(), password_sta.c_str());

  unsigned long startAttemptTime = millis();

  while (WiFi.status() != WL_CONNECTED && millis() - startAttemptTime < 10000) {

    delay(200);

    Serial.print(".");

  }

  if (WiFi.status() == WL_CONNECTED) {

    Serial.println("\nWiFi conectado");

    Serial.print("IP: ");

    Serial.println(WiFi.localIP());

    String msg = "WiFi conectado\nIP: " + WiFi.localIP().toString();

    displayMessage(msg.c_str());

  } else {

    Serial.println("\nFalha na conexao WiFi");

    displayMessage("Falha WiFi\nAcesse esp32_ap");

  }

}



void startAPMode() {

  Serial.println("Iniciando Access Point para configurar WiFi...");

  WiFi.mode(WIFI_AP);

  WiFi.softAP("esp32_ap");

  Serial.print("IP do AP: ");

  Serial.println(WiFi.softAPIP());

  displayMessage("Esp32 modo AP\nRede: esp32_ap");

}



void setupWebServer() {

  server.on("/", HTTP_GET, []() {

    String html = "<!DOCTYPE html><html><head><title>Config WiFi</title></head><body>";

    html += "<h1>Configurar WiFi</h1>";

    html += "<form action='/save' method='POST'>";

    html += "SSID: <input type='text' name='ssid' value='" + ssid_sta + "'><br>";

    html += "Senha: <input type='password' name='password' value='" + password_sta + "'><br>";

    html += "<input type='submit' value='Salvar'>";

    html += "</form></body></html>";

    server.send(200, "text/html", html);

  });

  server.on("/save", HTTP_POST, []() {

    if (server.hasArg("ssid") && server.hasArg("password")) {

      ssid_sta = server.arg("ssid");

      password_sta = server.arg("password");

      preferences.putString("ssid", ssid_sta);

      preferences.putString("password", password_sta);

      wifiConfigured = false; // Para tentar reconectar depois

      String resp = "<!DOCTYPE html><html><body>";

      resp += "<h1>Salvo! Reinicie o ESP para conectar.</h1>";

      resp += "</body></html>";

      server.send(200, "text/html", resp);

      Serial.println("WiFi configurado via web:");

      Serial.println("SSID: " + ssid_sta);

      Serial.println("Senha: " + password_sta);

    } else {

      server.send(400, "text/plain", "Requisição inválida");

    }

  });

  // Captura outras URLs e redireciona para "/"

  server.onNotFound([]() {

    server.sendHeader("Location", "/");

    server.send(302, "text/plain", "");

  });

  server.begin();

}

void readSensors() {

  // --- PH ---

  float phVoltage = readAverageADC(PH_SENSOR_PIN) * 3.3 / 4095.0;

  float phValue = voltageToPH(phVoltage);

  // --- TDS/EC ---

  float tdsRaw = readAverageADC(TDS_SENSOR_PIN);

  float ecValue = interpolateEC(tdsRaw);

  float ecComp = temperatureCompensateEC(ecValue, temperature_C);

  // --- DHT22 ---

  temperature_C = dht.readTemperature();

  humidity = dht.readHumidity();

  // --- DS18B20 ---

  ds18b20.requestTemperatures();

  water_temperature_C = ds18b20.getTempCByIndex(0);

  // --- SERIAL MONITOR ---

  Serial.print("pH: "); Serial.print(phValue,2);

  Serial.print(" | V: "); Serial.print(phVoltage,3);

  Serial.print(" | EC: "); Serial.print(ecComp,0);

  Serial.print(" | AirTemp: "); Serial.print(temperature_C,1);

  Serial.print(" | Hum: "); Serial.print(humidity,1);

  Serial.print(" | WaterTemp: "); Serial.println(water_temperature_C,1);

}



void updateDisplay() {

  display.clearDisplay();

  display.setTextSize(1);

  display.setCursor(0,0);

  float phVoltage = readAverageADC(PH_SENSOR_PIN) * 3.3 / 4095.0;

  float phValue = voltageToPH(phVoltage);

  float tdsRaw = readAverageADC(TDS_SENSOR_PIN);

  float ecValue = interpolateEC(tdsRaw);

  float ecComp = temperatureCompensateEC(ecValue, temperature_C);

  switch (currentPage) {

    case SHOW_PH:

      display.println("pH Value");

      display.setTextSize(2); display.setCursor(0,20); display.print(phValue,2);

      display.setTextSize(1); display.setCursor(0,50); display.print("V: "); display.print(phVoltage,3);

      display.setCursor(100,0); display.print("1/5");

      break;

    case SHOW_TDS:

      display.println("TDS/EC Value");

      display.setTextSize(2); display.setCursor(0,20); display.print(ecComp,0); display.print(" uS/cm");

      display.setTextSize(1); display.setCursor(0,50); display.print("Raw: "); display.print(tdsRaw,0);

      display.setCursor(100,0); display.print("2/5");

      break;

    case SHOW_AIR_TEMP:

      display.println("Air Temp & Hum");

      display.setTextSize(2); display.setCursor(0,20); display.print(temperature_C,1); display.print(" C");

      display.setTextSize(1); display.setCursor(0,50); display.print("Hum: "); display.print(humidity,1); display.print("%");

      display.setCursor(100,0); display.print("3/5");

      break;

    case SHOW_WATER_TEMP:

      display.println("Water Temp");

      display.setTextSize(2); display.setCursor(0,20); display.print(water_temperature_C,1); display.print(" C");

      display.setTextSize(1); display.setCursor(0,50);

      if (water_temperature_C == DEVICE_DISCONNECTED_C) display.print("Sensor erro");

      else display.print("DS18B20 OK");

      display.setCursor(100,0); display.print("4/5");

      break;

    case SHOW_HUMIDITY:

      display.println("Humidity");

      display.setTextSize(2); display.setCursor(0,20); display.print(humidity,1); display.print(" %");

      display.setTextSize(1); display.setCursor(0,50); display.print("AirTemp: "); display.print(temperature_C,1); display.print(" C");

      display.setCursor(100,0); display.print("5/5");

      break;

  }

  display.display();

}



float readAverageADC(int pin, int samples) {

  long sum = 0;

  for (int i=0;i<samples;i++){

    sum += analogRead(pin);

    delay(5);

  }

  return (float)sum/(float)samples;

}



float voltageToPH(float voltage) {

  float ph = ph_slope * voltage + ph_intercept;

  if(ph < 0) ph = 0;

  if(ph > 14) ph = 14;

  return ph;

}



float interpolateEC(float rawValue) {

  float slope = (calibration_high_ec - calibration_low_ec) / (calibration_high_raw - calibration_low_raw);

  float ecValue = calibration_low_ec + slope * (rawValue - calibration_low_raw);

  if(ecValue < 0) ecValue = 0;

  return ecValue;

}



float temperatureCompensateEC(float ec, float temp) {

  return ec / (1 + 0.02 * (temp - 25.0));

}



void calculatePHCoefficients() {

  float pH_low = 4.0;

  float pH_neutral = 7.0;

  ph_slope = (pH_neutral - pH_low) / (cal_ph7_voltage - cal_ph4_voltage);

  ph_intercept = pH_neutral - ph_slope * cal_ph7_voltage;

}

  preferences.putFloat("cal_ph7_voltage", cal_ph7_voltage);

  preferences.putFloat("cal_ph4_voltage", cal_ph4_voltage);

  preferences.putFloat("calibration_high_raw", calibration_high_raw);

  preferences.putFloat("calibration_low_ec", calibration_low_ec);

  preferences.putFloat("calibration_high_ec", calibration_high_ec);

  Serial.println("Calibracao salva.");

}

}



void handleButtons() {

  int buttonState[4] = {

    digitalRead(BUTTON_UP),

    digitalRead(BUTTON_DOWN),

    digitalRead(BUTTON_SELECT),

    digitalRead(BUTTON_BACK)

  };

  unsigned long currentMillis = millis();

  if(buttonState[0]==LOW && (currentMillis-lastDebounce[0])>debounceDelay){

    lastDebounce[0]=currentMillis;

    Serial.println("Botao UP");

    currentPage = (Page)((currentPage+1)%5);

  }

  if(buttonState[1]==LOW && (currentMillis-lastDebounce[1])>debounceDelay){

    lastDebounce[1]=currentMillis;

    Serial.println("Botao DOWN");

    currentPage = (Page)((currentPage+4)%5);

  }

  if(buttonState[2]==LOW && (currentMillis-lastDebounce[2])>debounceDelay){

    lastDebounce[2]=currentMillis;

    Serial.println("Botao SELECT - calibracao nao implementada");

  }

  if(buttonState[3]==LOW && (currentMillis-lastDebounce[3])>debounceDelay){

    lastDebounce[3]=currentMillis;

    Serial.println("Botao BACK - calibracao nao implementada");

  }

}



void displayMessage(const char *message) {

  display.clearDisplay();

  display.setTextSize(1);

  display.setCursor(0,0);

  display.println(message);

  display.display();

}
